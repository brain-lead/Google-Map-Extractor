For a standard search, it appears as:!1s{query}!7i20!10b1!12m6!1m1!18b1!2m1!20e3!6m1!114b1!17m1!3e1!20m57!...[cite: 9]!1s{query}: The URL-encoded search keyword (e.g., plumbers+in+austin).!7i20: The requested page size (instructs the server to return 20 results). import smtplib
import socket

def verify_email_smtp(email_address, sender_domain="my-daas-startup.com"):
    local_part, domain = email_address.rsplit('@', 1)
    
    # Step 1: Query MX records via DNS
    try:
        mx_records = dns.resolver.resolve(domain, 'MX')
        # Sort by preference priority
        mx_hosts = [r.exchange.to_text() for r in sorted(mx_records, key=lambda r: r.preference)]
    except Exception:
        return "undeliverable", "No MX Records Found"

    if not mx_hosts:
        return "undeliverable", "No Active MX Hosts"

    target_mx = mx_hosts[0]
    
    # Step 2: Establish Socket Connection and SMTP Session
    try:
        with smtplib.SMTP(host=target_mx, timeout=5.0) as smtp:
            smtp.helo(sender_domain)
            smtp.mail(f"verification-agent@{sender_domain}")
            
            # RCPT TO command asks if the target mailbox exists
            code, message = smtp.rcpt(email_address)
            
            # Response code 250/251 indicates a valid mailbox
            if code in (250, 251):
                return "verified", "Deliverable"
            # Response code 550 indicates the email does not exist
            elif code == 550:
                return "undeliverable", "Mailbox Does Not Exist"
            else:
                return "risky", f"SMTP Server returned response code: {code}"
    except (socket.error, smtplib.SMTPException) as e:
        return "unknown", f"Connection error occurred: {str(e)}"
